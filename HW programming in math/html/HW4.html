    y=x+y;
    x=y-x;
    n=n-1;
    end
    if mod(k,100)==11
        fprintf('the %dth term of fibonacci sequence is %d\n',k,y)

        fprintf('the %dth term of fibonacci sequence is %d\n',k,y)
