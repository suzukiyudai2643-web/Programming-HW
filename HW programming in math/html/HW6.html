Q2

f=@(x) cos(x)-x;
g=@(x) -sin(x)-1;

x1=0.74;
x2=0.74;
n=0;

while true
    x2=x1-(f(x1)/g(x1));
    n=n+1;
    plot(n,x2,'*')
    hold on

    if abs(x1-x2)<10^(-10)
        fprintf('the root found in %d interations: x=%f\n',n,x2)
        break
    end
    x1=x2;
end
