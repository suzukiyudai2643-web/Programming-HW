clear
clc
g=@(x) cos(x);

p0=0;
TOL=10^(-5);
N0=100;
i=1;
while i<=N0
    p=g(p0);
    plot(i,p,'.')
    hold on

    if abs(p-p0)<TOL
        fprintf('the procedure was successful: root is:%f \n',p)
        break
    end
    i=i+1;
    p0=p;
    if i==N0
        fprintf('the method failed after %d interations, the procedure was unsuccessful.\n ',i)
    end

end
title('the approximation of the root')
xlabel('the number of interation')
ylabel('the approximated value')
